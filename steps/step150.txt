import { Redirect, Stack } from 'expo-router'
import { useAuth } from '@clerk/clerk-expo'

export default function AuthRoutesLayout() {
  const { isSignedIn } = useAuth()

  if (isSignedIn) {
    return <Redirect href={'/'} />
  }

  return <Stack />
}

So, this checks basiclaly that a user is signned in or not ; if signned in > they will be directed to '/' i.e. the homepage ; else they will be shown the signup OR the signin page there.
